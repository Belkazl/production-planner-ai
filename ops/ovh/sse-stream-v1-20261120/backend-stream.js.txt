  // Each equipment lane shows its FIRST 10 OPERATIONS in chronological
  // order, including setup/changeover and production. A setup counts as
  // one of the ten, even without an order id. No arbitrary operation picking.
  const isSetup = op => op?.is_setup === true ||
    String(op?.phase||"").toLowerCase()==="setup" ||
    String(op?.type||"").toLowerCase()==="setup";
  const boundedPreview = (plan,model) => {
    if(!plan || plan.valid===false || !model)return null;
    const equipment=(model.equipment||model.autoclaves||[]).slice(0,14);
    if(!equipment.length)return null;
    const aliases=new Map();
    equipment.forEach((row,i)=>{
      for(const id of [row.id,row.name,row.__ppid])
        if(id!=null)aliases.set(String(id),i);
    });
    const ownOrders=new Set((model.orders||[]).map(x=>String(x.id)));
    const nativeOps=Array.isArray(plan.operations)?plan.operations:[];
    const genericOps=Array.isArray(plan.generic_operations)?plan.generic_operations:[];
    // Both representations are used by the real worker. Choose the one that
    // actually contains verified setup operations, not production-only native
    // rows that may omit them.
    const nativeSetups=nativeOps.filter(isSetup).length;
    const genericSetups=genericOps.filter(isSetup).length;
    const primary=genericSetups>nativeSetups?genericOps:
      nativeOps.length?nativeOps:genericOps;
    // The real GA adapter emits checked changeovers in a SEPARATE
    // `setup_operations` array. Both `operations` and `generic_operations`
    // can contain only production rows in an interim snapshot. Merge them
    // before choosing the first ten, never invent synthetic setup blocks.
    const equipmentIds=op=>Array.isArray(op.equipment_ids)&&op.equipment_ids.length?
      op.equipment_ids:op.equipment_id!=null?[op.equipment_id]:
      Array.isArray(op.resources)?op.resources
        .filter(r=>r?.resource_type==="equipment")
        .map(r=>r.id??r.resource_ref??r.name):[];
    const footprint=op=>[Number(op.start_min),Number(op.end_min),
      equipmentIds(op).map(String).sort().join(",")].join("|");
    const setupKeys=new Set(primary.filter(isSetup).map(footprint));
    const separated=Array.isArray(plan.setup_operations)?
      plan.setup_operations:[];
    const detached=[];
    for(const op of separated){
      if(!op||typeof op!=="object")continue;
      const key=footprint(op);
      if(setupKeys.has(key))continue;
      setupKeys.add(key);
      detached.push({...op,is_setup:true,phase:"setup"});
    }
    const operations=[...primary,...detached];
    if(!operations.length||!aliases.size)return null;
    const lanes=Array.from({length:equipment.length},()=>[]);
    for(const [sourceIndex,op] of operations.entries()){
      if(!op||typeof op!=="object")continue;
      const setup=isSetup(op);
      // A changeover may have no order id (or an internal batch id).
      // The job and its equipment lane are already account-scoped.
      if(!setup && (op.order==null || !ownOrders.has(String(op.order))))continue;
      const ids=equipmentIds(op);
      const targets=[...new Set(ids.map(id=>aliases.get(String(id))).filter(Number.isInteger))];
      if(!targets.length)continue;
      const start=Number(op.start_min),end=Number(op.end_min);
      if(!Number.isFinite(start)||!Number.isFinite(end)||end<=start)continue;
      for(const row of targets)lanes[row].push({op,sourceIndex,start,end,setup});
    }
    const chosen=new Map();
    for(const [row,entries] of lanes.entries()){
      // First ten operations in actual start-time order, NOT the first ten
      // raw records and NOT ten production jobs plus all their setups.
      // Setup precedes production on identical start times.
      entries.sort((a,b)=>a.start-b.start ||
        Number(b.setup)-Number(a.setup) || a.end-b.end || a.sourceIndex-b.sourceIndex);
      for(const entry of entries.slice(0,10)){
        let targets=chosen.get(entry.sourceIndex);
        if(!targets){targets=new Set();chosen.set(entry.sourceIndex,targets)}
        targets.add(row);
      }
    }
    if(!chosen.size)return null;
    const selected=[...chosen].map(([sourceIndex,targets])=>{
      const op=operations[sourceIndex];
      return {...op,equipment_ids:[...targets].sort((a,b)=>a-b).map(row=>
        String(equipment[row].id??equipment[row].__ppid??equipment[row].name))};
    }).sort((a,b)=>Number(a.start_min)-Number(b.start_min)||
      Number(isSetup(b))-Number(isSetup(a))||
      Number(a.end_min)-Number(b.end_min));
    const productionCount=selected.filter(op=>!isSetup(op)).length;
    const setupCount=selected.length-productionCount;
    // Use the VISIBLE first-ten-operation window for the interim time axis.
    // Keeping the complete optimization duration here compresses every real
    // changeover into a near-invisible sliver at the far left of the Gantt.
    // Full-candidate metrics still arrive separately in _pp_preview_kpis.
    const visibleEnd=Math.max(1,Math.ceil(Math.max(...selected.map(op=>Number(op.end_min)))));
    // At most 10 actual operations per lane, including setups. 14 equipment
    // lanes mean a maximum of 140 unique preview operations, not more.
    if(selected.length>equipment.length*10)
      throw Error("preview_more_than_ten_operations_per_lane");
    // KPI numbers belong to the COMPLETE verified candidate, never the
    // truncated preview used to paint the lightweight Gantt.
    const kpiKeys=["duration_minutes","batch_count",
      "operator_wait_minutes","machine_idle_minutes"];
    const values=Object.fromEntries(kpiKeys.map(key=>[key,Number(plan[key])]));
    const metrics=kpiKeys.every(key=>plan[key]!=null &&
      Number.isFinite(values[key])&&values[key]>=0)?values:null;
    return {
      valid:true,ok:true,partial:true,_pp_preview_limited:true,
      _pp_preview_rows:equipment.length,_pp_preview_per_row:10,
      _pp_preview_setups_included:true,
      _pp_preview_detached_setups:detached.length,
      _pp_preview_production_count:productionCount,
      _pp_preview_setup_count:setupCount,
      operations:selected,generic_operations:selected,batches:[],
      start:plan.start||model.start||null,
      duration_minutes:visibleEnd,
      _pp_preview_full_duration_minutes:plan.duration_minutes||null,
      _pp_preview_kpis:metrics
    };
  };
  async function stream(req){
    let principal,model,job=null;
    try{
      principal=await account(req);
      model=await readAccountModel(req);
    }catch{return response({ok:false,error:'sse_auth_or_model_unavailable'},503)}
    const modelKey=hashModel(model);
    const expectedId=new URL(req.url).searchParams.get('job_id');
    if(expectedId&&!/^[0-9a-f]{32}$/.test(expectedId))
      return response({ok:false,error:'invalid_job_id'},400);
    const encoder=new TextEncoder();
    let aborter=null,previewTimer=null,closed=false,working=false,previewBusy=false;
    let lastPreviewHash="",lastPercent=-1,lastPhase="",lastSeq=-1;
    const payload=(event,data)=>"event: "+event+"\ndata: "+JSON.stringify(data)+"\n\n";
    const body=new ReadableStream({
      start(controller){
        const emit=(event,data)=>{
          if(closed)return;
          try{controller.enqueue(encoder.encode(payload(event,data)))}
          catch(_){closed=true;aborter?.abort()}
        };
        const close=()=>{
          if(closed)return;
          closed=true;
          if(previewTimer)clearInterval(previewTimer);
          aborter?.abort();
          try{controller.close()}catch(_){}
        };
        const sendPreview=async()=>{
          if(closed||!working||previewBusy)return;
          previewBusy=true;
          try{
            const res=await progress(req);
            if(!res.ok)return;
            const data=await res.json();
            if(closed||!data?.running)return;
            const setupRows=p=>{
              if(!p||typeof p!=="object")return 0;
              const native=Array.isArray(p.operations)?p.operations:[];
              const generic=Array.isArray(p.generic_operations)?p.generic_operations:[];
              const detached=Array.isArray(p.setup_operations)?p.setup_operations:[];
              return Math.max(native.filter(isSetup).length,
                generic.filter(isSetup).length,detached.length);
            };
            // Never combine changeovers from one candidate with production
            // from another. If tentative candidate lacks its setup schedule,
            // show the independently verified setup-aware BEST plan instead.
            const live=data.candidate_preview,stored=data.best_preview;
            const useBest=setupRows(live)===0&&setupRows(stored)>0;
            const source=useBest?"best":live?"candidate":"best";
            const candidate=boundedPreview(useBest?stored:(live||stored),model);
            if(candidate){
              candidate._pp_preview_origin=source;
              const identity=candidate.operations.map(x=>[
                x.operation_id||x.id||"",x.order??"",isSetup(x)?"setup":"job",
                x.start_min,x.end_min,x.equipment_ids?.join(',')||""
              ].join(':')).join('|')+"|KPI:"+JSON.stringify(candidate._pp_preview_kpis);
              if(identity!==lastPreviewHash){
                lastPreviewHash=identity;
                emit("preview",{job_id:job.id,plan:candidate});
              }
            }
          }catch(_){}finally{previewBusy=false}
        };
        // Real server worker continuously pushes progress. Gantt is an
        // independent server-owned 4s preview, not a browser HTTP poll.
        const task=(async()=>{
          aborter=new AbortController();
          try{
            // Send the HTTP 200 streaming response once, before POST /api/plan
            // has finished creating its account job. This eliminates the 409
            // request storm during the initial optimization startup race.
            emit("ready",{transport:"one-sse-stream"});
            for(let tries=0;tries<120&&!closed;tries++){
              let candidate=null;
              try{candidate=await currentJob(principal)}catch(_){}
              const eligible=candidate&&candidate.modelKey===modelKey&&
                (expectedId?candidate.id===expectedId:
                  ["queued","running"].includes(candidate.state));
              if(eligible){job=candidate;break;}
              if(tries%12===11)emit("heartbeat",{waiting_for:"job"});
              await new Promise(resolve=>setTimeout(resolve,750));
            }
            if(closed)return;
            if(!job){emit("problem",{reason:"sse_job_start_timeout"});return;}
            if(job.state==="completed"&&job.result){
              const result=await progress(req);
              const state=await result.json();
              if(result.ok&&state?.success&&state?.phase==="completed")
                emit("complete",{job_id:job.id,percent:100,verified:true});
              else emit("error",{job_id:job.id,reason:"sse_terminal_unverified"});
              return;
            }
            const worker=await fetchImpl(origin+"/internal/jobs/"+job.id+"/events",{
              method:"GET",
              headers:{authorization:"Bearer "+token,"x-authenticated-account":principal,
                accept:"text/event-stream"},
              signal:aborter.signal,redirect:"manual",cache:"no-store"
            });
            if(!worker.ok||!worker.body)throw Error("worker_events_http_"+worker.status);
            const reader=worker.body.getReader(),decoder=new TextDecoder();
            working=true;
            previewTimer=setInterval(()=>{void sendPreview()},4000);
            emit("progress",{job_id:job.id,percent:Math.min(99,job.progress||0),
              phase:job.state,stage:"queued",seq:0});
            let buffer="";
            for(;;){
              const part=await reader.read();
              if(part.done)break;
              buffer+=decoder.decode(part.value,{stream:true}).replaceAll("\r","");
              if(buffer.length>32000)throw Error("worker_sse_frame_too_large");
              let at;
              while((at=buffer.indexOf("\n\n"))>=0){
                const frame=buffer.slice(0,at).replaceAll("\r","");
                buffer=buffer.slice(at+2);
                const dataLines=frame.split("\n").filter(x=>x.startsWith("data:"));
                if(!dataLines.length)continue;
                let data;
                try{data=JSON.parse(dataLines.map(x=>x.slice(5).trimStart()).join("\n"))}
                catch{continue}
                if(data.job_id!==job.id||!Number.isInteger(data.seq)||data.seq<=lastSeq)continue;
                lastSeq=data.seq;
                const phase=String(data.status||"running");
                const pct=Math.max(0,Math.min(99,Number(data.progress)||0));
                if(phase==="queued"||phase==="running"){
                  working=true;
                  if(pct!==lastPercent||phase!==lastPhase){
                    lastPercent=pct;lastPhase=phase;
                    emit("progress",{job_id:job.id,percent:pct,phase,stage:data.stage||"optimizing",
                      seq:lastSeq});
                  }
                }else{
                  working=false;
                  if(previewTimer){clearInterval(previewTimer);previewTimer=null}
                  if(phase==="completed"){
                    const checked=await progress(req);
                    const result=checked.ok?await checked.json().catch(()=>null):null;
                    if(checked.ok&&result?.success===true&&result?.phase==="completed"&&
                       result.result?.valid!==false&&result.result?.operations?.length){
                      emit("progress",{job_id:job.id,percent:100,phase:"completed",seq:lastSeq});
                      emit("complete",{job_id:job.id,percent:100,verified:true});
                    }else emit("error",{job_id:job.id,reason:"sse_final_validation_failed"});
                  }else emit("stopped",{job_id:job.id,phase});
                  return;
                }
              }
            }
            if(!closed)throw Error("worker_events_disconnected");
          }catch(error){
            if(!closed){
              const message=String(error?.message||"");
              const reason=aborter?.signal.aborted?"sse_aborted":
                /^worker_events_http_[0-9]{3}$/.test(message)?message:
                message==="worker_events_disconnected"?"worker_disconnected":
                error?.name==="TypeError"?"worker_fetch_type_error":
                "worker_event_stream_failed";
              emit("problem",{job_id:job?.id||null,reason});
            }
          }finally{close()}
        })();
        void task;
      },
      cancel(){closed=true;working=false;if(previewTimer)clearInterval(previewTimer);
        aborter?.abort();}
    });
    return new Response(body,{status:200,headers:{
      "content-type":"text/event-stream; charset=utf-8",
      "cache-control":"no-store, no-transform",
      "x-accel-buffering":"no",
      "x-content-type-options":"nosniff"
    }});
  }
